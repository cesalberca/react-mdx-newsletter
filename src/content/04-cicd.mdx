## The Send Newsletter Action

Runs daily at 07:00 UTC. If a newsletter file with today's date exists, it broadcasts it.

```yaml
on:
  schedule:
    - cron: '0 7 * * *'
  workflow_dispatch:

jobs:
  send-newsletter:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6

      - name: Find today's newsletter
        id: find
        run: |
          TODAY=$(date +%Y-%m-%d)
          FILE=$(find src/newsletter/emails/newsletter -name "${TODAY}_*.tsx" | head -1)
          if [ -z "$FILE" ]; then
            echo "newsletter_found=false" >> $GITHUB_OUTPUT
          else
            SLUG=$(basename "$FILE" .tsx)
            echo "newsletter_found=true" >> $GITHUB_OUTPUT
            echo "newsletter_slug=$SLUG" >> $GITHUB_OUTPUT
          fi

      - name: Broadcast
        if: steps.find.outputs.newsletter_found == 'true'
        run: |
          curl -sf -X POST ${{ vars.NEXT_PRODUCTION_URL }}/api/newsletter/broadcast \
            -H "Content-Type: application/json" \
            -d '{
              "newsletterSlug": "${{ steps.find.outputs.newsletter_slug }}",
              "token": "${{ secrets.NEWSLETTER_BROADCAST_TOKEN }}"
            }'
```

## The Broadcast API

```typescript
import { render } from "@react-email/render";

const newsletterModule = await import(
  `@/emails/newsletter/${newsletterSlug}.tsx`
);
const NewsletterComponent = newsletterModule.default;
const { title } = NewsletterComponent;

const html = await render(NewsletterComponent() as ReactElement);

const broadcast = await resend.broadcasts.create({
  segmentId: NEWSLETTER_CONFIG.SEGMENT_ID,
  topicId: NEWSLETTER_CONFIG.RESEND_TOPIC_ID,
  from: env.RESEND_EMAIL_FROM,
  subject: title,
  html,
  name: title,
});

await resend.broadcasts.send(broadcast.data.id, {
  scheduledAt: "today 15:00 UTC",
});
```

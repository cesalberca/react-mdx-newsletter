## The GitHub Action

```yaml
name: Send Newsletter
on:
  push:
    branches: [main]
    paths: ['emails/issues/**.mdx']

jobs:
  send:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - run: npm ci

      - name: Build email HTML
        run: npm run build:emails

      - name: Preview URL
        run: npm run deploy:preview
        env:
          VERCEL_TOKEN: $\{{ secrets.VERCEL_TOKEN }}

      - name: Send via Resend
        run: npm run send
        env:
          RESEND_API_KEY: $\{{ secrets.RESEND_API_KEY }}
```

## The Send Script

```typescript
// scripts/send.ts
import { Resend } from 'resend'
import { render } from '@react-email/render'
import { getLatestIssue } from '../lib/issues'
import { getSubscribers } from '../lib/subscribers'
import { WeeklyEmail } from '../emails/templates/weekly'

const resend = new Resend(process.env.RESEND_API_KEY)

async function main() {
  const issue = await getLatestIssue()
  const subscribers = await getSubscribers()

  const html = await render(
    <WeeklyEmail issue={issue} />
  )

  // Resend supports batch sending
  const { data, error } = await resend.batch.send(
    subscribers.map(sub => ({
      from: 'Newsletter <news@yourdomain.com>',
      to: sub.email,
      subject: issue.subject,
      html,
    }))
  )

  console.log(`Sent to ${data?.length} subscribers`)
  if (error) console.error('Errors:', error)
}

main()
```

## Resend API Basics

Setting up Resend is straightforward:

```typescript
import { Resend } from 'resend'

const resend = new Resend('re_123456789')

// Send a single email
await resend.emails.send({
  from: 'you@yourdomain.com',
  to: 'subscriber@example.com',
  subject: 'Welcome!',
  html: '<h1>Hello World</h1>',
})
```

### Key features:
- **Batch sending** — up to 100 emails per batch call
- **Webhooks** — get notified on delivery, bounce, complaint
- **Analytics** — open rates, click rates
- **Domain verification** — SPF, DKIM, DMARC setup
- **React Email integration** — first-class support
